what your phone does
// 1. Build the packet (minus signatures)
const packet = {
  claims: {
    audience:  "solesupply.com",
    issuer:    "device:iphone-15-pro",
    scope:     ["pay"],
    subject:   "user@icloud.com"
  },
  header: {
    expires_at: "",
    issued_at:  "",
    nonce:      "",
    packet_id:  "",
    tp_version: "1.0"
  },
  payload: {
    amount:   "189.00",
    currency: "USD",
    item:     "Jordan 4 Retro - Size 11",
    order_id: ""
  }
};

// 2. Canonicalize (RFC-2001, Section 5.3)
//    - sort all keys alphabetically
//    - strip whitespace
//    - UTF-8, no BOM
const canonical = canonicalize(packet);
canonicalized output (what gets hashed)
Every key is alphabetically sorted. All whitespace stripped. This is deterministic — any system in the world will produce the exact same string from the same packet.
hashing & signing
// 3. Hash the canonical form -> TPH
const tph = sha256(canonical);
// ""

// 4. Sign the TPH with device secure enclave
//    Private key NEVER leaves the chip
const signature = secureEnclave.sign(tph, {
  algorithm: "ed25519",
  key_id:    "2026q1-user",
  requireBiometric: true  // Face ID
});

// 5. Attach signature to packet
packet.signatures = [{
  algorithm: "ed25519",
  key_id:    "2026q1-user",
  signature: signature,
  signer:    "user@icloud.com"
}];
trust packet hash (TPH)
This hash is what gets signed. Change any field — even a single character in the item name — and the entire hash changes. Try it above.
signature
Created by the secure enclave after Face ID. This proves this device approved this exact purchase.

// Step 4: CREATE delegation packet
const delegation = {
  claims: {
    delegated_to: "stripe.com",
    issuer:       "solesupply.com",
    parent_tph:   tph,  // links to user's packet
    scope:        ["pay:max_189"],
    constraints:  { amount_max: 189.00 }
  },
  header: {
    expires_at: "",  // shorter!
    tp_version: "1.0"
  }
};

// Step 5: SIGN with store's key
const delTph = sha256(canonicalize(delegation));
delegation.signatures = [{
  signer: "solesupply.com",
  signature: store.sign(delTph)
}];

what stripe does
// Stripe receives BOTH packets
// Full chain verification (RFC-2001, Section 6)

function verifyChain(userPacket, delegation) {

  // 1. Verify user's signature
  const userKey = dns.lookupMultiResolver(
    "2026q1-user._domainkey.icloud.com",
    { resolvers: 3, requireConsensus: true }
  );  // RFC-3001: 3+ resolvers must agree
  assert(ed25519.verify(userKey, userTph, userSig));

  // 2. Verify store's signature
  const storeKey = dns.lookupMultiResolver(
    "2026q1._domainkey.solesupply.com",
    { resolvers: 3, requireConsensus: true }
  );
  assert(ed25519.verify(storeKey, delTph, storeSig));

  // 3. Check hash chain
  assert(delegation.claims.parent_tph === userTph);

  // 4. Check scope narrows (never expands)
  assert(isSubset(
    delegation.claims.scope, userPacket.claims.scope
  ));

  // 5. Check time narrows
  assert(
    delegation.header.expires_at <=
    userPacket.header.expires_at
  );

  // 6. Replay check
  assert(!seenBefore(userPacket.header.nonce));

  return "AUTHORIZED";
}
